const {chromium}=require('../gym-next/node_modules/@playwright/test');
const fs=require('node:fs');const path=require('node:path');
const dir=path.resolve(__dirname,'../docs/assets/links-netlify-2026-11-02');
(async()=>{const browser=await chromium.launch();const context=await browser.newContext({viewport:{width:1440,height:1100},reducedMotion:'no-preference'});const page=await context.newPage();const checks=[];
try{
await page.goto('https://getyourmentor-demo-20261014.netlify.app/',{waitUntil:'networkidle'});const consent=page.getByRole('button',{name:'Refuser le non nécessaire',exact:true});if(await consent.count())await consent.click();

const style=await page.locator('[data-mentor-pentagon]').evaluate(e=>({border:getComputedStyle(e).borderTopWidth,background:getComputedStyle(e).backgroundColor,overflow:document.documentElement.scrollWidth>innerWidth}));if(style.border!=='0px'||style.overflow)throw Error(JSON.stringify(style));

faces.push(await active.evaluate(e=>({id:e.querySelector('[data-mentor-id]')?.getAttribute('data-mentor-id'),imageCount:e.querySelectorAll('img').length,link:e.querySelector('a')?.getAttribute('href')})));

}finally{await context.close();await browser.close();}})().catch(e=>{console.error(e);process.exitCode=1;});
